1. npm i dotenv

2. Create .env

3. Choose endpoint - https://hapi.fhir.org/baseR4

-------------------------------------------------------
Return total number of medications based on relevance
-------------------------------------------------------
curl -s \
  -H 'Accept: application/fhir+json' \
  -H 'Cache-Control: no-store' \
  'https://hapi.fhir.org/baseR4/MedicationRequest?_count=5&_total=accurate' \
  | jq '{total, entries: (.entry | length), links: [.link[]?.relation]}'

-------------------------------------------------------
Result
-------------------------------------------------------
{
    "total": 29381,
    "entries": 0,
    "links": [
        "self"
    ]
}


-------------------------------------------------------
Find patients with multiple recent medication orders
-------------------------------------------------------
curl -s \
-H 'Accept: application/fhir+json' \
-H 'Cache-Control: no-store' \
'https://hapi.fhir.org/baseR4/MedicationRequest?_count=50&_sort=-_lastUpdated' \
| jq -r '.entry[]?.resource.subject.reference' | sort | uniq -c | sort -rn | head

Number of medication orders for speceific PT
-------------------------------------------------------
PT=137237825
curl -s -H 'Accept: application/fhir+json' -H 'Cache-Control: no-store' \
  "https://hapi.fhir.org/baseR4/MedicationRequest?patient=$PT&_count=50&_total=accurate" \
  | jq '{total, entries: (.entry | length)}'

-------------------------------------------------------

}


-------------------------------------------------------
Return MedicationRequest for a single resource
-------------------------------------------------------
curl -s -H 'Accept: application/fhir+json' \
  'https://hapi.fhir.org/baseR4/MedicationRequest?_count=1&_sort=-_lastUpdated' \
  | jq '.entry[0].resource
        | {id, status, intent, medicationCodeableConcept, medicationReference, contained, subject}'
